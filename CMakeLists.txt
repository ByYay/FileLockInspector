cmake_minimum_required(VERSION 3.24)
project(FileLockInspector VERSION 1.0.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 and MSVC with Windows 11 SDK are required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_compile_options(/W4 /permissive- /utf-8 /EHsc /guard:cf)
add_link_options(/DYNAMICBASE /NXCOMPAT /guard:cf)
add_library(fli_core STATIC src/core.cpp src/process.cpp src/scan_job.cpp)
target_include_directories(fli_core PUBLIC src)
target_link_libraries(fli_core PUBLIC rstrtmgr advapi32 shell32 ole32 user32 psapi)
add_executable(FileLockInspector WIN32 src/main.cpp src/ui.cpp resources/app.rc)
target_link_libraries(FileLockInspector PRIVATE fli_core comctl32 dwmapi uxtheme)
target_include_directories(FileLockInspector PRIVATE resources)
target_link_options(FileLockInspector PRIVATE /MANIFEST:NO)
add_executable(FileLockWorker src/worker.cpp resources/app.rc)
target_link_libraries(FileLockWorker PRIVATE fli_core)
target_include_directories(FileLockWorker PRIVATE resources)
target_link_options(FileLockWorker PRIVATE /MANIFEST:NO)
add_library(FileLockShell SHARED shell/ExplorerCommand.cpp shell/exports.def)
target_include_directories(FileLockShell PRIVATE src)
target_link_libraries(FileLockShell PRIVATE shell32 shlwapi ole32 uuid)
include(CTest)
if(BUILD_TESTING)
  add_executable(fli_holder tests/holder.cpp)
  target_include_directories(fli_holder PRIVATE src)
  add_executable(fli_tests tests/tests.cpp)
  target_link_libraries(fli_tests PRIVATE fli_core)
  add_dependencies(fli_tests FileLockWorker)
  add_test(NAME core_integration COMMAND fli_tests)
  set_tests_properties(core_integration PROPERTIES TIMEOUT 120)
endif()
install(TARGETS FileLockInspector FileLockWorker FileLockShell RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(DIRECTORY installer/ DESTINATION installer)
install(FILES README.md LICENSE DESTINATION .)
